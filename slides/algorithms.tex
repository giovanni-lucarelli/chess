\section{RL Algorithms}

\begin{frame}{Value Iteration: model-based approach}
  \begin{itemize}
  \item KRk: 182.676 states
  \item Feasible number of states + perfect knowledge $\rightarrow$ Dynamic Programming
  \item $V(s) \leftarrow \max_a \sum_{s'}p(s'|s,a)[r(s,a,s')+\gamma V(s')]$
  \item $\pi(s) = \arg\max_a \sum_{s'}p(s'|s,a)[r(s,a,s')+\gamma V(s')]$
  \end{itemize}
\end{frame}

\begin{frame}{Choice of parameters}
  \begin{itemize}
    \item \textbf{Step penalty}: -2 $\rightarrow$ values = DTZ
    \item \textbf{Draw penalty}: -100 $\rightarrow$ slow mates > quick draws
    \item \textbf{Checkmate reward}: +1
    \item $\mathbf{\gamma} = 1$
    \item $\mathbf{\theta} = 0.01$
  \end{itemize}
\end{frame}

\begin{frame}{Q-Learning: model-free approach}
  \begin{itemize}
    \item How good of a policy without knowing the ``level" of the opponent?
    \item $Q(s,a) \leftarrow Q(s,a) + \alpha[r + \gamma \max_{a'}Q(s',a') - Q(s,a)]$
    \item $\pi(s) = \arg\max_a Q(s,a)$
    \item Off-policy, online and one-step.
  \end{itemize}
\end{frame}

\begin{frame}{Choice of parameters}
  \begin{itemize}
    \item \textbf{Max steps} per episode: 50 $\rightarrow$ if it's too far from mate, agent may never reach terminal state
    \item \textbf{Step/Draw/Mate}: -0.01/-1/+1
    \item $\mathbf{\gamma} = 0.99$
    \item Costant \textbf{learning rate} $\alpha = 0.05$ $\rightarrow$ smaller variance
    \item $\mathbf{\varepsilon}$-greedy policy with $\varepsilon = 0.3$ and  $\varepsilon_{\textit{decay}}=0.99$
    \item $\varepsilon_t = 0.3\varepsilon_{\textit{decay}}^{t+1}$
  \end{itemize}
\end{frame}

\begin{frame}{Actor-critic: approximate solution approach}
  \begin{itemize}
    \item Non-elementary chess endgames $\rightarrow$ \textbf{curse of dimensionality}
    \item Use one-step actor-critic
    \item Update rules:
    \begin{itemize}
      \item $\delta = r + \gamma \hat{V}(s';w) - \hat{V}(s;w)$
      \item $\theta \leftarrow \theta + \alpha^{\theta} \delta \nabla_{{\theta}} \log \pi(a|s;\theta) $
      \item $w \leftarrow w + \alpha^{w} \delta \nabla_w \hat{V}(s;w)$
    \end{itemize} 
  \end{itemize}
\end{frame}

\begin{frame}{Parameters Choice}
\begin{itemize}
    \item \textbf{Policy}: MLP(128,128,64), ReLU activations, softmax output.
    \begin{itemize}
      \item \textbf{Input}: pieces coordinates + stm (7 inputs) 
    \end{itemize}
    \item \textbf{Value function}: $\hat{V}(s;w) = \sum_{i=1}^{16} w_i \phi_i(s)$
    \item $\gamma = 0.99$
    \item \textbf{Step/Draw/Mate}: -0.01/-1/+1
    \item $\alpha^w = 0.02$
    \item $\alpha^\theta = 0.003$
    \item Too many iterations and no generalization $\rightarrow$ curriculum learning
\end{itemize}
\end{frame}

\begin{frame}{Monte Carlo Tree Search}
  \begin{itemize}
    \item One of the most common approaches to chess
    \item No need to train!
    \item \textbf{Checkmate reward}: +1
    \item \textbf{Draw penalty}: -1
    \item \textbf{Seconds per move}: $10$s
    \item \textbf{Exploration constant}: $c=\sqrt{2}$
    \item \textbf{Upper confidence bound} applied to trees:
    $UCT(s,a) = Q(s,a) + c \sqrt{\frac{\ln N(s)}{N(s,a)}}$
  \end{itemize}
\end{frame}
