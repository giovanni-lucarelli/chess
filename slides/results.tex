\begin{frame}{Critic through human chess principles}
    \centering
    \begin{itemize}
        \item \texttt{d\_edgeBK} = -0.269
        \item \texttt{d\_cornerBK} = -0.402
        \item \texttt{dInf(WK,BK)} = -0.286
        \item \texttt{dMan(WK,BK)} = -0.375
        \item \texttt{dInf(WR,BK)} = -0.069
        \item \texttt{dMan(WR,BK)} = +0.200
        \item \texttt{cut\_w} = +0.103
    \end{itemize}
\end{frame}
